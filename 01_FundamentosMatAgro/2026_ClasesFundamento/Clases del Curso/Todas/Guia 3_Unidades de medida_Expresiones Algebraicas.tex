	\section*{III. Problemas contextualizados}
	
	\begin{enumerate}[itemsep=0.5cm]
		\item Un sistema de riego entrega $1200\ \text{mL/min}$ de agua.  
		¿A cuántos litros por hora equivale esta tasa de riego?
		
		\item Un estanque agrícola tiene forma de prisma rectangular de:
		\begin{itemize}
			\item Largo: $4\ \text{m}$
			\item Ancho: $2\ \text{m}$
			\item Profundidad: $1.5\ \text{m}$
		\end{itemize}
		¿Cuántos litros de agua se necesitan para llenarlo completamente?
		
		\item En una plantación se utilizan:
		\begin{itemize}
			\item 3 sacos de fertilizante de $25\ \text{kg}$ cada uno
			\item 5 sacos de compost de $18\ \text{kg}$ cada uno
		\end{itemize}
		¿Cuál es la masa total utilizada en toneladas?
		
		\item Un tractor recorre $15\ \text{km}$ en $30\ \text{min}$.
		\begin{itemize}
			\item ¿Cuál es su velocidad en km/h?
			\item ¿Cuál es su velocidad en m/s?
		\end{itemize}
		
		\item Un cultivo requiere $2.5\ \text{L}$ de agua por planta al día.  
		Si hay 350 plantas:
		\begin{itemize}
			\item ¿Cuántos litros se necesitan diariamente?
			\item ¿Cuántos metros cúbicos representa ese volumen?
		\end{itemize}
	\end{enumerate}

	\section*{V. Problemas aplicados}
		
	\begin{enumerate}
		
		\item \textbf{Costo total:}
		
			\begin{itemize}
				\item Costo Producción: $A = 3x^2 	+ 5xy - 2xy^2$
				\item Costo Fertilización: $B = 10xy + 7x^2 - 3y^2$
				\item Costo Plagas: $C = 4y^2 - 8xy + 10$
			\end{itemize}
			
			\textbf{Pregunta:}
			
			¿Cuál es el costo total?
			
		\item \textbf{Crecimientos vegetales:}
					
			$A = 2x^3 - x^2 + 4x - 5$
			
			$B = x^3 - 8x - 2$
			
			$C = x^2 - 4x - 2$
			
			\textbf{Pregunta:} 
		
			¿Cuál es el crecimiento vegetal: $4A - [2B + C - (3B + 2A)] +10B$?
		
		\item \textbf{Fertilización en un cultivo:}
			
			\begin{itemize}
				\item Base: $(2n + 15)$
				\item Refuerzo: $(3n - 10)$
			\end{itemize}
			
			\textbf{Pregunta:}
			 
			¿Cuál es la cantidad total aplicada?
	 
		\item \textbf{Nutrientes:}
		
			\begin{itemize}
				\item Suelo: $45x^2 - 12xy + 30y^2$
				\item Fertilizante: $-7x^2 + 24xy - 3y^2$
				\item Abono: $30x^2 - 10xy + y^2$
			\end{itemize}
			
			\textbf{Preguntas:}
			
			\begin{itemize}
				\item Con abono
				\item Sin abono
			\end{itemize}
			
		\item \textbf{Pesticidas:}
			
			Disponible: $50a + 20b + 30$
			
			Aplicación:
			
			\begin{itemize}
				\item Parcela 1: $5a - 35b + 42$
				\item Parcela 2: $32a + 10b - 22$
			\end{itemize}
			
			\textbf{Pregunta:}  
			¿Cuánto queda?
			
	\end{enumerate}
