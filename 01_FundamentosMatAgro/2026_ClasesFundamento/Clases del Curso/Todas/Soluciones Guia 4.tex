	\centering \section*{Soluciones Guía 4}
	
\begin{enumerate}
	
	\item \textbf{Simplifique las siguientes expresiones algebraicas:}
	\begin{enumerate}[label=\alph*)]
		\item $120mn - 54mn + 100m - 30mn - 28m = 36mn + 72m$
		\item $35pq + 30 + 78pq - 60pq - 13 = 53pq + 17$
		\item $14 + [12xy - (3x^2y^2 - 4xy) + 8xy + 10] = -3x^2y^2 + 24xy + 24$
		\item $-10t + [-(-2t + 3) + 12] - (14 + 2t + 11t) = -21t - 5$
		\item $x^2yz - (18xy^2z + 10xyz^2) = x^2yz - 18xy^2z - 10xyz^2$
		\item $11a^2b - 32a^2b + 14ab^2 - 3a^2b = -24a^2b + 14ab^2$
		\item $10x^2yz - 12x^2yz - 8yx^2z + 42yx^2z = 32x^2yz$
		\item $54rt^2 - (12rt^2 - 5rt) + (21rt^2 - 2rt) = 63rt^2 + 3rt$
	\end{enumerate}
	
	\item \textbf{Costo total:}
	\[
	A+B+C=(3x^2+5xy-2xy^2)+(10xy+7x^2-3y^2)+(4y^2-8xy+10)
	\]
	\[
	=10x^2+7xy-2xy^2+y^2+10
	\]
	
	\item \textbf{Cantidad total aplicada:}
	\[
	(2n+15)+(3n-10)=5n+5
	\]
	
	\item \textbf{Nutrientes:}
	
	\textbf{Con abono:}
	\[
	(45x^2-12xy+30y^2)+(-7x^2+24xy-3y^2)+(30x^2-10xy+y^2)
	\]
	\[
	=68x^2+2xy+28y^2
	\]
	
	\textbf{Sin abono:}
	\[
	(45x^2-12xy+30y^2)+(-7x^2+24xy-3y^2)=38x^2+12xy+27y^2
	\]
	
	\item \textbf{Crecimientos vegetales:}
	\[
	4A-[2B+C-(3B+2A)]+10B
	\]
	Primero,
	\[
	2B+C-(3B+2A)=-B+C-2A
	\]
	Entonces,
	\[
	4A-[-B+C-2A]+10B=6A+11B-C
	\]
	Sustituyendo:
	\[
	6(2x^3-x^2+4x-5)+11(x^3-8x-2)-(x^2-4x-2)
	\]
	\[
	=23x^3-7x^2-60x-50
	\]
	
	\item \textbf{Pesticidas:}
	\[
	(50a+20b+30)-[(5a-35b+42)+(32a+10b-22)]
	\]
	\[
	=(50a+20b+30)-(37a-25b+20)=13a+45b+10
	\]
	
	\item \textbf{Charlas técnicas:}
	\begin{enumerate}[label=\alph*)]
		\item 
		\[
		4xy-(x-2y)^2=4xy-(x^2-4xy+4y^2)=-x^2+8xy-4y^2
		\]
		\[
		(4x+y)(4y+x)=4x^2+17xy+4y^2
		\]
		\[
		(x-2y)(x+2y)-(2x-y)^2=(x^2-4y^2)-(4x^2-4xy+y^2)
		\]
		\[
		=-3x^2+4xy-5y^2
		\]
		
		\item Operaciones:
		\begin{itemize}
			\item Primera: trinomio cuadrado y reducción.
			\item Segunda: producto de binomios.
			\item Tercera: suma por diferencia y cuadrado de binomio.
		\end{itemize}
		
		\item La charla con mayor expresión algebraica simplificada es:
		\[
		4x^2+17xy+4y^2
		\]
		
		\item Sí, hay expresiones más complejas de lo necesario, porque deben simplificarse para compararse correctamente.
	\end{enumerate}
	
	\item \textbf{Costos de análisis:}
	\begin{enumerate}[label=\alph*)]
		\item
		\[
		(9a+7b)(a+4b)=9a^2+43ab+28b^2
		\]
		\[
		(a+2b)(2a-3b)=2a^2+ab-6b^2
		\]
		\[
		(2a-b)(3a+14b)=6a^2+25ab-14b^2
		\]
		
		\item Costos individuales:
		\[
		2a^2+ab-6b^2 \quad \text{y} \quad 6a^2+25ab-14b^2
		\]
		
		\item Suma de costos individuales:
		\[
		(2a^2+ab-6b^2)+(6a^2+25ab-14b^2)=8a^2+26ab-20b^2
		\]
		No coincide con
		\[
		9a^2+43ab+28b^2
		\]
		
		\item Sí, el modelo total es inconsistente con la suma de los dos análisis dados.
	\end{enumerate}
	
	\item \textbf{Producción total:}
	\[
	P(x,y)=(x+y)^2+(x-y)^2+2xy
	\]
	\[
	=(x^2+2xy+y^2)+(x^2-2xy+y^2)+2xy
	\]
	\[
	=2x^2+2y^2+2xy
	\]
	Interpretación: combina producción base en $x$ e $y$ con interacción $2xy$.
	
	\item \textbf{Propuesta del técnico:}
	\[
	(x+3)^2=x^2+6x+9
	\]
	No es correcto. El error está en que
	\[
	2\cdot x\cdot 3=6x,\quad \text{no } 9x
	\]
	
	\item \textbf{Error en el modelo:}
	\[
	(x+4)^2=x^2+8x+16
	\]
	Por tanto,
	\[
	x^2+6x+8\neq (x+4)^2
	\]
	
	\item \textbf{Modelo total de fertilización intensiva:}
	\begin{enumerate}[label=\alph*)]
		\item
		\[
		(2n+15)(3n-10)+(n+5)^2
		\]
		\[
		=6n^2+25n-150+n^2+10n+25
		\]
		
		\item
		\[
		=7n^2+35n-125
		\]
		
		\item El término $7n^2$ domina el crecimiento del modelo; $35n$ representa crecimiento lineal y $-125$ es ajuste constante.
		
		\item Cuando $n$ aumenta linealmente, el modelo crece cuadráticamente.
		
		\item No hay redundancia luego de simplificar; los términos semejantes ya quedaron reducidos.
	\end{enumerate}
	
	\item \textbf{Factorizaciones:}
	\begin{enumerate}[label=\alph*)]
		\item $6x^3y+9x^2y=3x^2y(2x+3)$
		\item $8x^2-12x=4x(2x-3)$
		\item $4x^2-49=(2x-7)(2x+7)$
		\item $x^4-81=(x^2-9)(x^2+9)=(x-3)(x+3)(x^2+9)$
		\item $9a^2-16b^2=(3a-4b)(3a+4b)$
		\item $x^2+10x+25=(x+5)^2$
		\item $9x^2+12xy+4y^2=(3x+2y)^2$
		\item $x^2+7x+12=(x+3)(x+4)$
		\item $x^2+9x+20=(x+4)(x+5)$
		\item $x^2+5x+6=(x+2)(x+3)$
	\end{enumerate}
	
	\item \textbf{Rendimiento del cultivo:}
	\[
	R(x)=x^2+11x+30=(x+5)(x+6)
	\]
	
	Si se escribe como
	\[
	(x+11)(x+30)
	\]
	entonces:
	\[
	(x+11)(x+30)=x^2+41x+330
	\]
	que no coincide con el modelo original.
	
\end{enumerate}
