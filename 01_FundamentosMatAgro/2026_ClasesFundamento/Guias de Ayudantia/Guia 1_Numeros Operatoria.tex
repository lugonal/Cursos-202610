			\small Números y Operatoria}
	\end{center}
	
	\vspace{1em}
	
	\section*{Parte I: Conjuntos numéricos}
	
	Determine el conjunto numérico mínimo al que pertenece cada resultado
	
	\begin{enumerate}
		\item $17 + 23 + 13 + 7$
		\item $(12 + 18) + (8 + 2)$
		\item $6(15 + 5)$
		\item $8(20 + 10) - 3(20 + 10)$
		\item $5(14 + 6) + 4(14 + 6)$
		\item $7(12 + 8) + 5(12 + 8) - 3(12 + 8)$
	\end{enumerate}
	
	Indique si el resultado pertenece a: $\mathbb{N}, \mathbb{Z}, \mathbb{Q}, \mathbb{R}$. Justifique.
	
	% =========================
	\section*{Parte II: Cierre de operaciones}
	
	Analice si cada operación está cerrada en el conjunto indicado.
	
	\begin{enumerate}
		\item $8 - 13$ en $\mathbb{N}$
		\item $15 \div 5$ en $\mathbb{Z}$
		\item $\frac{2}{3} + \frac{5}{6}$ en $\mathbb{Q}$
		\item $\sqrt{5} + 3$ en $\mathbb{R}$
		\item $7 - 12$ en $\mathbb{Z}$
	\end{enumerate}
	
	Explique en cada caso.

	\section*{Parte IV: Análisis de errores}
	
	Un estudiante afirma:
	
	\[
	(14 - 6)\cdot 3 = 14 - (6 \cdot 3)
	\]
	
	\begin{enumerate}
		\item Calcule ambos lados de la igualdad.
		\item Determine si la igualdad es correcta.
		\item Identifique el error conceptual.
		\item Escriba la propiedad correcta.
	\end{enumerate}

	\section*{Parte V: Balance hídrico}
	
	Durante una semana se registran los siguientes datos:
	
	\begin{itemize}
		\item Lluvia: $+18$ mm
		\item Riego: $+11$ mm
		\item Evapotranspiración: $-23$ mm
		\item Escurrimiento: $-4$ mm
	\end{itemize}
	
	\begin{enumerate}
		\item Calcule el balance hídrico total.
		\item Reescriba la operación agrupando entradas y salidas.
		\item Interprete el resultado.
	\end{enumerate}

	\section*{Parte VI: Fertilización}
	
	Un cultivo requiere una dosis de:
	
	\[
	150 \text{ kg/ha}
	\]
	
	La superficie cultivada es:
	
	\[
	3{,}8 \text{ ha}
	\]
	
	\begin{enumerate}
		\item Calcule la dosis total.
		\item Si cada saco contiene $25$ kg, determine cuántos sacos se necesitan.
		\item Analice si el resultado requiere redondeo.
	\end{enumerate}

	\section*{Parte IX: Validación}
	
	Un técnico agrícola está calculando la cantidad de fertilizante que debe aplicarse en un cultivo de maíz.
	
	La recomendación técnica indica aplicar:
	
	\[
	140 \text{ kg/ha}
	\]
	
	La superficie sembrada es de:
	
	\[
	3{,}6 \text{ ha}
	\]
	
	El técnico realiza el siguiente cálculo:
	
	\[
	140 \times 3{,}6 = 504
	\]
	
	\textbf{Preguntas}
	
	\begin{enumerate}
		\item Verifique el cálculo realizado por el técnico. ¿Es correcto el resultado?
		\item Determine si el resultado es razonable considerando el problema planteado.
		\item Analice el resultado considerando:
		\begin{itemize}
			\item Unidad: ¿Qué unidad debe tener el resultado final?
			\item Magnitud: ¿Es razonable para $3{,}6$ ha?
			\item Coherencia del contexto
		\end{itemize}
		\item Si el fertilizante se vende en sacos de $25$ kg:
		\begin{itemize}
			\item ¿Cuántos sacos deben comprarse?
			\item ¿Se debe redondear? Justifique.
		\end{itemize}
	\end{enumerate}
