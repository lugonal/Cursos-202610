\section*{Pauta de soluciones}

\begin{enumerate}[label=\textbf{\arabic*.}]
	
	%================================================
	\item 
	\[
	180x+240=1320
	\]
	\[
	180x=1320-240
	\]
	\[
	180x=1080
	\]
	\[
	x=\frac{1080}{180}=6
	\]
	
	\textbf{Respuesta:} El terreno posee \(6\) hectáreas.
	
	%================================================
	\item 
	\[
	25x+1250=3500
	\]
	\[
	25x=3500-1250
	\]
	\[
	25x=2250
	\]
	\[
	x=\frac{2250}{25}=90
	\]
	
	\textbf{Respuesta:} Faltan \(90\) cajas de tomates.
	
	%================================================
	\item 
	\[
	18x+240=960
	\]
	\[
	18x=720
	\]
	\[
	x=\frac{720}{18}=40
	\]
	
	\textbf{Respuesta:} Son \(40\) surcos de cultivo.
	
	%================================================
	\item 
	\[
	45x-120=780
	\]
	\[
	45x=900
	\]
	\[
	x=\frac{900}{45}=20
	\]
	
	\textbf{Respuesta:} Se requieren \(20\) sacos de semilla.
	
	%================================================
	\item 
	\[
	12(x+5)=180
	\]
	\[
	x+5=\frac{180}{12}
	\]
	\[
	x+5=15
	\]
	\[
	x=10
	\]
	
	\textbf{Respuesta:} Hay \(10\) sectores de riego.
	
	%================================================
	\newpage
	\vspace*{0.2cm}
	\item 
	\[
	\frac{x}{4}+35=95
	\]
	\[
	\frac{x}{4}=60
	\]
	\[
	x=60\cdot 4
	\]
	\[
	x=240
	\]
	
	\textbf{Respuesta:} Se distribuyen \(240\) litros de agua.
	
	%================================================
	\item 
	\[
	\frac{2x}{5}-18=42
	\]
	\[
	\frac{2x}{5}=60
	\]
	\[
	2x=300
	\]
	\[
	x=150
	\]
	
	\textbf{Respuesta:} El rendimiento es \(150\) kg por parcela.
	
	%================================================
	\item 
	\[
	\frac{x+30}{3}=70
	\]
	\[
	x+30=210
	\]
	\[
	x=180
	\]
	
	\textbf{Respuesta:} La producción es de \(180\) plantas injertadas.
	
	%================================================
	\item 
	\[
	P=15x+120
	\]
	\[
	P-120=15x
	\]
	\[
	x=\frac{P-120}{15}
	\]
	
	\textbf{Respuesta:}
	\[
	\boxed{x=\frac{P-120}{15}}
	\]
	
	%================================================
	\item 
	\[
	C=rx+350
	\]
	\[
	C-350=rx
	\]
	\[
	r=\frac{C-350}{x}
	\]
	
	\textbf{Respuesta:}
	\[
	\boxed{r=\frac{C-350}{x}}
	\]
	
	%================================================
	\newpage
	\vspace*{0.2cm}
	\item 
	\[
	T=\frac{q}{4}+25
	\]
	\[
	T-25=\frac{q}{4}
	\]
	\[
	q=4(T-25)
	\]
	
	\textbf{Respuesta:}
	\[
	\boxed{q=4(T-25)}
	\]
	
	%================================================
	\item 
	Cada planta requiere \(2,8\) litros diarios. Del estanque de \(980\) litros se reservan \(140\) litros.
	
	\[
	2,8x+140=980
	\]
	\[
	2,8x=840
	\]
	\[
	x=\frac{840}{2,8}
	\]
	\[
	x=300
	\]
	
	\textbf{Respuesta:} Se pueden regar \(300\) plantas.
	
	\textbf{Validación:}
	\[
	300\cdot 2,8=840
	\]
	y al sumar los \(140\) litros reservados:
	\[
	840+140=980
	\]
	
	%================================================
	\item 
	\[
	P=ax+500
	\]
	
	Despejamos \(x\):
	
	\[
	P-500=ax
	\]
	\[
	x=\frac{P-500}{a}
	\]
	
	Si \(P=3200\) y \(a=45\):
	
	\[
	x=\frac{3200-500}{45}
	\]
	\[
	x=\frac{2700}{45}
	\]
	\[
	x=60
	\]
	
	\textbf{Respuesta:} Se cultivan \(60\) hectáreas.
	
	%================================================
	\newpage
	\vspace*{0.2cm}
	\item 
	\[
	350000+85000x=1200000
	\]
	\[
	85000x=1200000-350000
	\]
	\[
	85000x=850000
	\]
	\[
	x=\frac{850000}{85000}
	\]
	\[
	x=10
	\]
	
	\textbf{Respuesta:} Debe pagar \(10\) cuotas.
	
	\textbf{Consistencia práctica:} El resultado es entero, por lo tanto no requiere redondeo.
	
	%================================================
	\item 
	Resolvemos cada ecuación:
	
	\[
	10x+100=200
	\]
	\[
	10x=100
	\]
	\[
	x=10
	\]
	
	\[
	20x+100=300
	\]
	\[
	20x=200
	\]
	\[
	x=10
	\]
	
	\[
	30x+100=400
	\]
	\[
	30x=300
	\]
	\[
	x=10
	\]
	
	\[
	40x+100=500
	\]
	\[
	40x=400
	\]
	\[
	x=10
	\]
	
	\textbf{Patrón detectado:} En todos los casos \(x=10\).
	
	\textbf{Regularidad:} La ecuación mantiene la estructura:
	
	\[
	kx+100=100+10k
	\]
	
	Por lo tanto:
	
	\[
	kx=10k
	\]
	
	\[
	x=10
	\]
	
	%================================================
	\newpage
	\vspace*{0.2cm}
	\item 
	\[
	15x+60=0
	\]
	\[
	15x=-60
	\]
	\[
	x=-4
	\]
	
	\textbf{Respuesta algebraica:} El cálculo es correcto.
	
	\textbf{Respuesta contextual:} No es válido si \(x\) representa número de sacos de fertilizante.
	
	\textbf{Inconsistencia:} No existen \(-4\) sacos en un contexto real de compra o aplicación agrícola.
	
	%================================================
	\item 
	La afirmación:
	
	\begin{quote}
		``Si una ecuación tiene solución negativa, entonces siempre está mal resuelta.''
	\end{quote}
	
	es \textbf{falsa}.
	
	\textbf{Ejemplo agrícola:}
	
	Supongamos que \(x\) representa la variación del nivel de agua de un estanque, en litros.
	
	\[
	x+120=80
	\]
	\[
	x=80-120
	\]
	\[
	x=-40
	\]
	
	\textbf{Interpretación:} El estanque disminuyó en \(40\) litros.
	
	En este caso, la solución negativa sí tiene sentido, porque representa una pérdida o disminución.
	
\end{enumerate}
